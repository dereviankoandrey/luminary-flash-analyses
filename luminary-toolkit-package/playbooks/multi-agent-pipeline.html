2. Pipeline Script (Python)

This script orchestrates all four agents using Ollama's API. Save as underwriter_pipeline.py.

#!/usr/bin/env python3
"""Luminary Multi-Agent Underwriting Pipeline v1.0

Orchestrates: Data Extractor → Underwriter → Risk Auditor → Report Writer
Requires: Ollama running locally (ollama serve)

Usage:
    python underwriter_pipeline.py --address "123 Main St, Phoenix AZ 85001"
    python underwriter_pipeline.py --json deal_inputs.json
    python underwriter_pipeline.py --scan deals_batch.csv
import json
import subprocess
import sys
from pathlib import Path
from datetime import datetime

OLLAMA_BASE = "http://localhost:11434"

# Agent model assignments
MODELS = {
    "extractor":  "mistral:7b",       # Fast, good at extraction
    "underwriter": "qwen2.5:14b",     # Strong reasoning for math
    "auditor":    "llama3:8b-instruct-q4_K_M",  # Rule-following strength  
    "reporter":   "qwen2.5:14b"       # Formatting + synthesis
}

SYSTEM_PROMPTS = {
    "extractor": """You are a real estate data extraction agent. Extract structured property data from the input below.

RULES:
- Output ONLY valid JSON matching the schema provided
- If a field is not found, use null (not estimation)
- Property type must be one of: single_family, duplex, triplex, quadplex, multi_family, commercial
- All monetary values in USD as integers
- All percentages as decimals (0.05 for 5%)

OUTPUT SCHEMA:
{
  "address": "string",
  "property_type": "enum",
  "bedrooms": int|null,
  "bathrooms": float|null,
  "square_feet": int|null,
  "lot_size_sqft": int|null,
  "year_built": int|null,
  "asking_price": int|null,
  "estimated_rent_low": int|null,
  "estimated_rent_high": int|null,
  "property_taxes_annual": int|null,
  "hoa_fees_monthly": int|null,
  "source": "string"
}

INPUT: {input_data}""",

    "underwriter": """You are a deterministic real estate underwriting analyst.

RULES: Show every calculation step. Never fabricate data. Flag gaps with DATA_GAP:.
DEFAULT ASSUMPTIONS (use if not specified): Vacancy=5%, Maintenance=8% of gross rent, 
Management=10%, Capex=3% of gross rent, Insurance=$1200/year multi-family/$800/year single-family.

INPUT DATA: {input_data}
MKT DATA: {market_data}

Calculate and output JSON with all metrics.""",

    "auditor": """You are a risk auditor for real estate underwriting analysis.

RISK CHECKLIST:
1. Cap rate within ±2% of market median? 
2. DSCR > 1.0? (FAIL if <1.0, WARN if 1.0-1.15)
3. Cash-on-cash return positive? (FAIL if negative)
4. All required data fields present?

INPUT ANALYSIS: {input_data}
AUDIT RULES: {audit_rules}

Output JSON with risk_score (0-100), critical_issues[], warnings[], passes[], recommendation.""",

    "reporter": """You are a professional real estate report writer. Convert the analysis data into a clean, 
professional underwriting report suitable for investor review.

FORMAT REQUIREMENTS:
- Use markdown formatting throughout
- Include all sections: Deal Summary, Income Analysis, Expense Breakdown, Cash Flow, Key Metrics, Risk Assessment
- Highlight critical warnings in bold
- Add a one-paragraph executive summary at the top
- End with clear recommendation (APPROVE/CONDITIONAL/REJECT) and reasoning

INPUT DATA: {input_data}
AUDIT RESULTS: {audit_results}"""
}
# Pipeline execution engine
class UnderwritingPipeline:
    def __init__(self):
        self.results = {}
        
    def run_agent(self, agent_name, system_prompt, user_input):
        """Execute a single agent via Ollama API with temperature control"""
        cmd = [
            "ollama", "run", MODELS[agent_name],
            f"{system_prompt}\n\nINPUT: {user_input}"
        ]
        result = subprocess.run(cmd, capture_output=True, text=True, timeout=120)
        return result.stdout

    def extract_data(self, raw_input):
        """Agent 1: Extract structured data from listing/address"""
        prompt = SYSTEM_PROMPTS["extractor"].format(input_data=raw_input)
        output = self.run_agent("extractor", prompt, "")
        
        # Parse JSON from Ollama response (may include markdown formatting)
        import re
        json_match = re.search(r'\{.*\}', output, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
        raise ValueError(f"Failed to extract structured data. Raw output:\n{output}")

    def underwrite(self, extracted_data, market_data=None):
        """Agent 2: Perform deterministic underwriting calculations"""
        prompt = SYSTEM_PROMPTS["underwriter"].format(
            input_data=json.dumps(extracted_data, indent=2),
            market_data=json.dumps(market_data or {}, indent=2)
        )
        output = self.run_agent("underwriter", prompt, "")
        
        json_match = re.search(r'\{.*\}', output, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
        raise ValueError(f"Underwriting analysis failed. Raw:\n{output}")

    def audit(self, deal_analysis):
        """Agent 3: Risk assessment via checklist"""
        prompt = SYSTEM_PROMPTS["auditor"].format(
            input_data=json.dumps(deal_analysis, indent=2),
            audit_rules="Standard risk thresholds per Luminary playbook"
        )
        output = self.run_agent("auditor", prompt, "")
        
        json_match = re.search(r'\{.*\}', output, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
        raise ValueError(f"Audit failed. Raw:\n{output}")

    def generate_report(self, deal_analysis, audit_results):
        """Agent 4: Professional report generation"""
        prompt = SYSTEM_PROMPTS["reporter"].format(
            input_data=json.dumps(deal_analysis, indent=2),
            audit_results=json.dumps(audit_results, indent=2)
        )
        return self.run_agent("reporter", prompt, "")

    def run(self, raw_input):
        """Execute full pipeline"""
        print("[Pipeline] Starting underwriting analysis...")
        
        # Stage 1: Data extraction
        print("[Stage 1/4] Extracting property data...")
        extracted = self.extract_data(raw_input)
        self.results["extracted_data"] = extracted
        
        # Stage 2: Underwriting analysis  
        print("[Stage 2/4] Performing underwriting calculations...")
        analysis = self.underwrite(extracted)
        self.results["analysis"] = analysis
        
        # Stage 3: Risk audit
        print("[Stage 3/4] Running risk audit...")
        audit = self.audit(analysis)
        self.results["audit"] = audit
        
        # Stage 4: Report generation
        print("[Stage 4/4] Generating report...")
        report = self.generate_report(analysis, audit)
        
        # Save results
        output_dir = Path("output") / datetime.now().strftime("%Y%m%d_%H%M%S")
        output_dir.mkdir(parents=True, exist_ok=True)
        
        for name, data in self.results.items():
            with open(output_dir / f"{name}.json", "w") as f:
                json.dump(data, f, indent=2)
        
        report_path = output_dir / "report.md"
        with open(report_path, "w") as f:
            f.write(report)
        
        print(f"[Pipeline] Complete. Results saved to {output_dir}")
        print(f"[Pipeline] Recommendation: {audit.get('recommendation', 'UNKNOWN')}")
        return report

# CLI entry point
if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="Luminary Underwriting Pipeline")
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--address", help="Property address to analyze")
    group.add_argument("--json", dest="json_file", help="JSON input file with deal data")
    group.add_argument("--scan", help="CSV batch of properties to scan")
    
    args = parser.parse_args()
    pipeline = UnderwritingPipeline()
    
    if args.address:
        pipeline.run(args.address)
    elif args.json_file:
        with open(args.json_file) as f:
            data = json.load(f)
        raw_input = json.dumps(data, indent=2)
        pipeline.run(raw_input)
